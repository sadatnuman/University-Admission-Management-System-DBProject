-- 1. CREATE MAIN USER

CREATE USER UAMSP
IDENTIFIED BY "1234"

GRANT UNLIMITED TABLESPACE TO UAMSP;



-- 2. BASIC LOGIN PRIVILEGES

GRANT CREATE SESSION TO UAMSP;



-- 3. DEVELOPMENT PRIVILEGES

GRANT
    CREATE TABLE,
    CREATE VIEW,
    CREATE SEQUENCE,
    CREATE PROCEDURE,
    CREATE TRIGGER,
    CREATE TYPE,
    CREATE SYNONYM,
    CREATE MATERIALIZED VIEW
TO UAMSP;



-- 4. ADVANCED DATA ACCESS

GRANT
    SELECT ANY TABLE,
    INSERT ANY TABLE,
    UPDATE ANY TABLE,
    DELETE ANY TABLE,
    ALTER ANY TABLE,
    DROP ANY TABLE
TO UAMSP;




-- 5. CREATE READ-ONLY ROLE

CREATE ROLE Numan;




-- 6. GRANT SELECT PRIVILEGES (UAMSP TABLES)

GRANT SELECT ON UAMSP.Admin       TO Numan;
GRANT SELECT ON UAMSP.Applicant   TO Numan;
GRANT SELECT ON UAMSP.Department  TO Numan;
GRANT SELECT ON UAMSP.Program     TO Numan;
GRANT SELECT ON UAMSP.Teacher     TO Numan;
GRANT SELECT ON UAMSP.Application TO Numan;
GRANT SELECT ON UAMSP.Student     TO Numan;
GRANT SELECT ON UAMSP.Payment     TO Numan;
GRANT SELECT ON UAMSP.Evaluation  TO Numan;



-- 7. ASSIGN ROLE TO USER
GRANT Numan TO UAMSP;
